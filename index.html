    Save the Zoom .csv to Google Drive (your backend copy of every upload)

    After the tool counts a file, it can send the original .csv to Drive through your engine workbook link, named date - file name.csv. Uploading the same day and file name again replaces the old copy.

    One-time setup in the engine script 
    // 1) In your engine script's doPost, next to the other actions, add:
//      case 'saveCsv': return <same JSON reply helper you already use>(saveCsv_(body));
// 2) Paste these two functions anywhere in the same project:
function saveCsv_(body) {
  var folder = getCsvFolder_(body.folderId);
  var name = String(body.date || 'undated') + ' - ' + String(body.fileName || 'zoom.csv').replace(/[\\\/:*?"<>|]/g, '_');
  var old = folder.getFilesByName(name);
  while (old.hasNext()) old.next().setTrashed(true);      // same day + same name = replace
  var f = folder.createFile(name, String(body.csv || ''), MimeType.CSV);
  return { ok: true, id: f.getId(), name: f.getName(), url: f.getUrl(), folder: folder.getName() };
}
function getCsvFolder_(id) {
  if (id) return DriveApp.getFolderById(id);
  var it = DriveApp.getFoldersByName('LBD Zoom CSV uploads');
  return it.hasNext() ? it.next() : DriveApp.createFolder('LBD Zoom CSV uploads');
}
// 3) Deploy > Manage deployments > Edit > New version > Deploy, and approve the Drive permission.
